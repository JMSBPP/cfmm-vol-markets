```plank
// PHASE 2 (VORD-01): VolOrder is a comptime TYPE CONSTRUCTOR VolOrder(T), following the in-repo
// Shock(R) template. T is a REGION TAG -- a phantom type parameter used as a region index
// (Haskell: the `s` in `ST s` / `runST`): it says WHERE the tokenId builder is guaranteed to find
// the (poolId, OptionRatio[4]) operands (maintainer's TODO.md note), and nothing else. The packed
// 248-bit word is region-agnostic: `extra` NEVER enters pack_vol_order / unpack_vol_order, which is
// what keeps this refactor's regression surface at zero for every packed-word consumer.
//
//   VolOrder(none)              -- the MINIMAL instantiation: today's four fields, NO `extra` field.
//                                  Reading `.extra` on it is a missing-field COMPILE error, never a
//                                  runtime check. Every current caller uses this one.
//   VolOrder(memory|calldata)   -- carries `extra: bytes(T)`, a {ptr,length} view into region T
//                                  (Phase 3 defines the payload layout; Phase 4 the wire format).
//   anything else               -- @compile_error, like region_ptr_type in std/regions.plk.
//
// `none` is declared HERE rather than adopting upstream `ctime` (std/regions.plk @ 00c0a1a): the
// semantics are the same ("no region"), but bytes(ctime) collapses to cbytes -- a comptime-only
// type -- and carrying that as a runtime struct field is an unverified corner of the compiler.
// Omitting the field is the strongest possible "cannot reach extra".
const none = struct {};

const VolOrder = fn (comptime T: type) type {
    if T == none {
        return struct {
            rangeWidth: VolRangeWidth,
            volStrike: TickVolatility,
            skew: SpreadTickAssimetry,
            targetVega: VegaTarget
        };
    }
    if T == memory or T == calldata {
        return struct {
            rangeWidth: VolRangeWidth,
            volStrike: TickVolatility,
            skew: SpreadTickAssimetry,
            targetVega: VegaTarget,
            extra: bytes(T)
        };
    }
    @compile_error("VolOrder: T must be none, memory or calldata");
};

// The ONE region-agnostic projection: the four packed fields of any VolOrder(T), as VolOrder(none).
// Identity for T == none. The generic tokenId path reads its geometry through this, so the only
// T-dependent read anywhere is (from Phase 3) `extra`. Mirrors shock_load_word being Shock's single
// region-branching site.
const vol_order_base = fn (comptime T: type, vo: VolOrder(T)) VolOrder(none) {
    if T == none { return vo; }
    VolOrder(none) {
        rangeWidth: vo.rangeWidth,
        volStrike: vo.volStrike,
        skew: vo.skew,
        targetVega: vo.targetVega
    }
};
```
