  1. check_cheat_tick t                            -- G4, client-side, BEFORE any RPC

  5. W_low  <- eth_call PriceSetterHook.packSlot0For(t)     -- the audited on-chain TickMath
  6. compose_slot0 W_now W_low                     -- bits 0..183 from W_low, >=184 from W_now
  7. anvil_setStorageAt(PoolManager, slot, composed)        -- does NOT mine a block (measured)
  8. evm_setNextBlockTimestamp(ts)                          -- ABSOLUTE. never evm_increaseTime.
  9. sendTransaction from DEPLOYER to swapRouter with encode_swap's calldata
 10. receiptStatus == 1, else FAIL LOUD with the tx hash
 11. exactly ONE E3 from changeAddress == DynamicFeeHook with topic1 == poolId
