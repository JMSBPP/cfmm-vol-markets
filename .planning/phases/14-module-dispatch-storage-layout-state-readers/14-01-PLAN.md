import std::constructor::return_runtime;
import interfaces::exposure::VegaAccountInterface::*;
import lib::exposure::VegaIssuanceLib::{haircut_risk_price, issue_shares};
import types::exposure::VegaExposure::{RiskPriceX96, Haircut};
import v3::util::{return_u256, revert_empty};
import std::error::require;

// FOUR distinct keccak-derived scalar slots. The preimage strings are RESTATED test-side so
// vm.load addresses are computable in Solidity (14-02). Each holds a plain u256 (no packing).
// "VegaAccountMod.totalDeposits"
const SLOT_TOTAL_DEPOSITS = 0x...;
// "VegaAccountMod.totalShares"
const SLOT_TOTAL_SHARES = 0x...;
// "VegaAccountMod.riskWeightedShares"
const SLOT_RISK_WEIGHTED_SHARES = 0x...;
// "VegaAccountMod.riskPrice"
const SLOT_RISK_PRICE = 0x...;

init { return_runtime(); }

run {
    let selector = @evm_shr(224, @evm_calldataload(0));

    if selector == SELECTOR_DEPOSIT {
        // deposit(uint256 collateralAmt) -- mints issue_shares(collateral, storedRiskPrice),
        // moves all three accumulators (d == 1: riskWeightedShares tracks totalShares in v1).
        let collateral = @evm_calldataload(4);

        // Guard 1: deposit(0) reverts. @evm_iszero, NOT @evm_not.
        if @evm_iszero(collateral) { revert_empty(); }

        // Guard 2 (unset price): storedRiskPrice != 0 means "set at least once" ONLY because
        // setRiskPrice REJECTS 0 (see SELECTOR_SET_RISK_PRICE below). DOCUMENTED COUPLING:
        // deleting the setter's zero-guard silently changes THIS check's meaning.
        let storedRiskPrice = @evm_sload(SLOT_RISK_PRICE);
        if @evm_iszero(storedRiskPrice) { revert_empty(); }

        // ALL issuance math routes through the lib -- the module holds ZERO arithmetic.
        let shares = issue_shares(collateral, RiskPriceX96 { val: storedRiskPrice });

        // Guard 3 (dust): sharesMinted == 0 reverts -- collateral cannot be banked for zero shares.
        if @evm_iszero(shares) { revert_empty(); }

        // Three distinct accumulators, checked adds.
        let new_total_deposits = @evm_sload(SLOT_TOTAL_DEPOSITS) + collateral;
        @evm_sstore(SLOT_TOTAL_DEPOSITS, new_total_deposits);
        @evm_sstore(SLOT_TOTAL_SHARES, @evm_sload(SLOT_TOTAL_SHARES) + shares);
        @evm_sstore(SLOT_RISK_WEIGHTED_SHARES, @evm_sload(SLOT_RISK_WEIGHTED_SHARES) + shares);

        // Admissibility guard (Flow.lean Q_M^Sigma), collapsed money-side form, Q_M^Sigma
        // instantiated as the POST-deposit totalDeposits. INERT SCAFFOLDING for the deferred flow
        // milestone: in deposit-only v1 new_total_deposits = old + collateral >= collateral ALWAYS
        // (collateral > 0 by Guard 1), so this can NEVER fire. It is NOT a live safety property;
        // its presence/form is verified solely by the 14-02 cross-product mutation. No boundary-
        // match test (that would be vacuous).
        require(new_total_deposits >= collateral);
        @evm_stop();

    } else if selector == SELECTOR_SET_RISK_PRICE {
        // setRiskPrice(uint256 pRiskX96). Stores the exogenous price; reverts on 0.
        // DELIBERATELY UNAUTHENTICATED in v1 -- scope boundary: the vault holds no custody, mints
        // nothing transferable, has no redemption path; authorization arrives with oracle wiring.
        let p = @evm_calldataload(4);
        if @evm_iszero(p) { revert_empty(); }
        @evm_sstore(SLOT_RISK_PRICE, p);
        @evm_stop();

    } else if selector == SELECTOR_PREVIEW_DEPOSIT {
        // previewDeposit(uint256 amt) -> uint256. Pure issuance via the SAME lib call as deposit,
        // no state change -- so preview can never diverge from the action (canonical vault bug).
        let amt = @evm_calldataload(4);
        let storedRiskPrice = @evm_sload(SLOT_RISK_PRICE);
        return_u256(issue_shares(amt, RiskPriceX96 { val: storedRiskPrice }));

    } else if selector == SELECTOR_PREVIEW_RISK_PRICE {
        // previewRiskPrice(uint256 oracleX96, uint256 hX96) -> uint256. The H1 computation with
        // h < 1 enforced on-chain via the lib's haircut_risk_price (its reverts come free).
        let oracleX96 = @evm_calldataload(4);
        let h = Haircut { val: @evm_calldataload(36) };
        return_u256(haircut_risk_price(oracleX96, h).val);

    } else if selector == SELECTOR_TOTAL_DEPOSITS {
        return_u256(@evm_sload(SLOT_TOTAL_DEPOSITS));

    } else if selector == SELECTOR_TOTAL_SHARES {
        return_u256(@evm_sload(SLOT_TOTAL_SHARES));

    } else if selector == SELECTOR_RISK_WEIGHTED_SHARES {
        return_u256(@evm_sload(SLOT_RISK_WEIGHTED_SHARES));

    } else if selector == SELECTOR_RISK_PRICE {
        return_u256(@evm_sload(SLOT_RISK_PRICE));
    }

    // Unknown selector: no fallback behaviour is defined for this module.
    revert_empty();
}
